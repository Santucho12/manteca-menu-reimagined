## Review Focus

1. Un mes que empieza (día 1) con el mes anterior más corto o más largo: los números comparan
   contra los mismos días, y el 31 contra el mes anterior entero (lo resuelve `periodoAnterior`
   de Reportes; Tarea 3 lo usa sin reimplementarlo).
2. Un local sin ningún cierre todavía: no se avisa "Ayer no se cargó el cierre" (Tarea 2).
3. Avisos de meses anteriores: el enlace lleva la fecha más vieja para que el filtro del mes no
   los esconda (Tareas 1 y 2).
4. Un cierre revisado, anulado o esperando ventas no cuenta como "con diferencia sin revisar";
   un gasto anulado no cuenta como sin clasificar (Tarea 2).
5. Un mes sin ventas: "Todavía no hay ventas este mes." y los números en cero sin romper

```ts
import { textoDePendientes } from '../usuarios/presentacion';

export type ConFecha = { cantidad: number; desde: string | null };
export type DatosDeAvisos = {
  conDiferencia: ConFecha;
  ayerSinCierre: string | null;
  pendientes: number;
  sinClasificar: ConFecha & { categoriaId: string | null };
};
export type Aviso = { texto: string; enlace: string };

const enPlural = (n: number, uno: string, varios: string) => (n === 1 ? uno : varios);

function deCierres({ cantidad, desde }: ConFecha): Aviso[] {
  if (cantidad === 0) return [];
  const que = enPlural(cantidad, 'cierre', 'cierres');
  return [
    {
      texto: `Hay ${cantidad} ${que} con diferencia sin revisar.`,
      enlace: `/admin/caja/control?estado=con_diferencia&desde=${desde}`,
    },
  ];
}

function deAyer(ayer: string | null): Aviso[] {
  if (!ayer) return [];
  return [
    {
      texto: 'Ayer no se cargó el cierre de caja.',
      enlace: `/admin/caja/control?desde=${ayer}&hasta=${ayer}`,
    },
  ];
}

const dePendientes = (n: number): Aviso[] =>
  n === 0 ? [] : [{ texto: textoDePendientes(n), enlace: '/admin/usuarios?estado=pendientes' }];

function deGastos({ cantidad, desde, categoriaId }: DatosDeAvisos['sinClasificar']): Aviso[] {
  if (cantidad === 0) return [];
  const que = enPlural(cantidad, 'gasto', 'gastos');
  return [
    {
      texto: `Hay ${cantidad} ${que} sin clasificar.`,
      enlace: `/admin/gastos?categoria=${categoriaId}&desde=${desde}`,
    },
  ];
}

/** Lo que requiere accion, en orden; cada aviso lleva a donde se resuelve. */
export function avisosDeInicio(datos: DatosDeAvisos): Aviso[] {
  return [
    ...deCierres(datos.conDiferencia),
    ...deAyer(datos.ayerSinCierre),
    ...dePendientes(datos.pendientes),
    ...deGastos(datos.sinClasificar),
  ];
}
```

```ts
import { sumarMeses } from '../gastos/vencimientos';
import type { Periodo } from '../reportes/periodos';
import { SIN_VINCULAR, type ProductoDelReporte } from '../reportes/ventas';
import { cantidadParaMostrar } from '../ventas/renglones';

const primeroDelMes = (fecha: string) => `${fecha.slice(0, 8)}01`;

export const mesEnCurso = (hoy: string): Periodo => ({ desde: primeroDelMes(hoy), hasta: hoy });

/** El grafico mes a mes: este mes y los 11 anteriores. */
export const ultimosDoceMeses = (hoy: string): Periodo => ({
  desde: sumarMeses(primeroDelMes(hoy), -11),
  hasta: hoy,
});

/** Los 5 que mas plata dejaron; lo no vinculado no es un producto y no entra. */
export const masVendidos = (productos: ProductoDelReporte[]) =>
  productos.filter((p) => p.clave !== SIN_VINCULAR).slice(0, 5);

export const textoDePresentaciones = (producto: ProductoDelReporte) =>
  producto.presentaciones.map((p) => `${cantidadParaMostrar(p.cantidad)} ${p.nombre}`).join(' · ');
```

```ts
import { diasConCierre, primerDiaConCierre } from '../caja/consultas';
import { datosDelResultado, ventasPorProducto } from '../reportes/consultas';
import { periodoAnterior, sumarDias } from '../reportes/periodos';
import { resultadoPorMes } from '../reportes/resultado';
import { productosDelReporte } from '../reportes/ventas';
import type { Cliente } from '../supabase/cliente';
import { contarPendientes } from '../usuarios/consultas';
import { avisosDeInicio, type ConFecha, type DatosDeAvisos } from './avisos';
import { masVendidos, mesEnCurso, ultimosDoceMeses } from './meses';

type Respuesta = { data: { fecha: string }[] | null; count: number | null; error: { message: string } | null };

function conFecha(r: Respuesta): ConFecha {
  if (r.error) throw new Error(`No pudimos leer los avisos: ${r.error.message}`);
  return { cantidad: r.count ?? 0, desde: r.data?.[0]?.fecha ?? null };
}

// Un cierre anulado o revisado ya no esta en con_diferencia: el estado alcanza.
async function cierresConDiferencia(cliente: Cliente): Promise<ConFecha> {
  return conFecha(
    await cliente
      .from('cierres_de_caja')
      .select('fecha', { count: 'exact' })
      .eq('estado', 'con_diferencia')
      .order('fecha')
      .limit(1),
  );
}

async function gastosSinClasificar(cliente: Cliente): Promise<DatosDeAvisos['sinClasificar']> {
  const categoria = await cliente.from('categorias_de_gasto').select('id').eq('de_sistema', true).maybeSingle();
  if (categoria.error) throw new Error(`No pudimos leer los avisos: ${categoria.error.message}`);
  const categoriaId = categoria.data?.id ?? null;
  if (!categoriaId) return { cantidad: 0, desde: null, categoriaId };
  const gastos = await cliente
    .from('gastos')
    .select('fecha', { count: 'exact' })
    .eq('categoria_id', categoriaId)
    .eq('anulado', false)
    .order('fecha')
    .limit(1);
  return { ...conFecha(gastos), categoriaId };
}

/** Ayer, si no tiene cierre; nada si el local todavia no cargo ninguno. */
async function ayerSinCierre(cliente: Cliente, hoy: string): Promise<string | null> {
  const ayer = sumarDias(hoy, -1);
  const [primero, cargados] = await Promise.all([primerDiaConCierre(cliente), diasConCierre(cliente, ayer, ayer)]);
  return !primero || primero > ayer || cargados.length > 0 ? null : ayer;
}

export async function datosDeAvisos(cliente: Cliente, hoy: string): Promise<DatosDeAvisos> {
  const [conDiferencia, ayer, pendientes, sinClasificar] = await Promise.all([
    cierresConDiferencia(cliente),
    ayerSinCierre(cliente, hoy),
    contarPendientes(cliente),
    gastosSinClasificar(cliente),
  ]);
  return { conDiferencia, ayerSinCierre: ayer, pendientes, sinClasificar };
}

/** Todo lo de Inicio, con el mismo calculo que Reportes. */
export async function datosDelInicio(cliente: Cliente, hoy: string) {
  const mes = mesEnCurso(hoy);
  const [avisos, actual, anterior, anio, productos] = await Promise.all([
    datosDeAvisos(cliente, hoy),
    datosDelResultado(cliente, mes),
    datosDelResultado(cliente, periodoAnterior(mes)),
    datosDelResultado(cliente, ultimosDoceMeses(hoy)),
    ventasPorProducto(cliente, mes),
  ]);
  return {
    avisos: avisosDeInicio(avisos),
    actual,
    anterior,
    meses: resultadoPorMes(anio.ventas, anio.gastos),
    masVendidos: masVendidos(productosDelReporte(productos)),
  };
}
```
